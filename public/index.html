async function loadLiveScore() { const updated = document.getElementById("scoreUpdated"); const list = document.getElementById("scoreList"); updated.innerHTML = "⏳ Live scores loading..."; list.innerHTML = ""; try { const response = await fetch("/api/live-score", { method: "GET", cache: "no-store" }); let data; try { data = await response.json(); } catch (e) { throw new Error("Live score server response invalid"); } if (!response.ok) { throw new Error( data.error || "Live score unavailable" ); } if ( !data.matches || !Array.isArray(data.matches) ) { throw new Error("No matches found"); } /* पुराने matches हटाएँ। आज और कल के आसपास के matches रखें। */ const today = new Date(); today.setHours(0, 0, 0, 0); const matches = data.matches.filter(match => { if (!match.date) { return true; } const matchDate = new Date(match.date); if (Number.isNaN(matchDate.getTime())) { return true; } matchDate.setHours(0, 0, 0, 0); const difference = (today - matchDate) / (1000 * 60 * 60 * 24); return difference <= 1; }); /* LIVE → STUMPS → DELAYED → UPCOMING → RESULT India matches को ऊपर दिखाएँ। */ matches.sort((a, b) => { function priority(match) { const status = String(match.status || "").toUpperCase(); const title = String(match.title || "").toLowerCase(); const teams = Array.isArray(match.teams) ? match.teams : []; let score = 50; if (status === "LIVE") { score = 1; } else if (status === "STUMPS") { score = 2; } else if (status === "DELAYED") { score = 3; } else if (status === "UPCOMING") { score = 4; } else if ( status === "RESULT" || status === "COMPLETED" ) { score = 5; } const indiaInTitle = title.includes("india"); const indiaInTeams = teams.some(team => String(team.name || "") .toLowerCase() .includes("india") ); if (indiaInTitle || indiaInTeams) { score -= 0.5; } return score; } return priority(a) - priority(b); }); updated.innerHTML = "Last updated: " + new Date().toLocaleTimeString() + " • " + matches.length + " matches • Auto refresh: 5 min"; /* कोई current match नहीं */ if (matches.length === 0) { list.innerHTML = `
🏏 अभी API में आज का कोई Live/Upcoming cricket match उपलब्ध नहीं है।

🔄 कुछ मिनट बाद Refresh दबाकर दोबारा check करें।
`; return; } /* Match cards */ matches.forEach(match => { const card = document.createElement("div"); card.className = "score-card"; const status = String( match.status || "LIVE" ).toUpperCase(); let statusClass = ""; if (status === "LIVE") { statusClass = "live"; } else if ( status === "RESULT" || status === "COMPLETED" ) { statusClass = "result"; } else if ( status === "DELAYED" || status === "RAIN DELAY" ) { statusClass = "delay"; } const teams = Array.isArray(match.teams) ? match.teams : []; const title = String( match.title || "Cricket Match" ); const indiaMatch = title.toLowerCase().includes("india") || teams.some(team => String(team.name || "") .toLowerCase() .includes("india") ); const indiaBadge = indiaMatch ? ` 🇮🇳 INDIA MATCH ` : ""; card.innerHTML = `
${escapeHTML(title)} ${indiaBadge}
${escapeHTML(status)}
${escapeHTML( match.description || "" )}
${teams.map(team => `
${escapeHTML( team.name || "" )}
${ team.overs && team.overs !== "-" ? "Overs: " + escapeHTML(team.overs) : "" }
${escapeHTML( team.score || "-" )}
`).join("")}
📍 ${escapeHTML( match.venue || "Cricket Ground" )}
📅 ${escapeHTML( match.date || "" )}
`; list.appendChild(card); }); } catch (error) { console.error( "Live Score Error:", error ); updated.innerHTML = ""; list.innerHTML = `
❌ ${escapeHTML( error.message )}

Live Score API से data प्राप्त नहीं हो सका।

🔄 Refresh दबाकर फिर कोशिश करें।
`; } }
